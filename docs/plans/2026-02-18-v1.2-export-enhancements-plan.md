### Task 3: Filename Template Resolver

**Files:**

- Modify: `index.html` (Script)

**Step 1: Implement `resolveTemplate` function**

```javascript
function resolveTemplate(template, data) {
  return template
    .replace(/{name}/g, data.name)
    .replace(/{w}/g, Math.round(data.w))
    .replace(/{h}/g, Math.round(data.h))
    .replace(/{idx}/g, data.idx);
}
```

### Task 5: Individual File Download

**Files:**

- Modify: `index.html` (Script)

**Step 1: Implement `downloadActiveFile` function**

```javascript
async function downloadActiveFile() {
  const f = files[activeIndex];
  if (!f) return;
  const crop = f.crop ?? batchCrop;
  if (!crop.w || !crop.h) {
    statusText.textContent = 'Select a crop region first.';
    return;
  }

  const format = document.getElementById('export-format').value;
  const ext = format === 'image/png' ? 'png' : format === 'image/jpeg' ? 'jpg' : 'webp';

  statusText.textContent = `Processing ${f.file.name}...`;

  const blob = await new Promise((resolve) => {
    const img = new Image();
    const url = URL.createObjectURL(f.file);
    img.onload = () => {
      const canvas = document.createElement('canvas');
      canvas.width = Math.round(crop.w);
      canvas.height = Math.round(crop.h);
      const ctx = canvas.getContext('2d');
      ctx.drawImage(
        img,
        Math.round(crop.x),
        Math.round(crop.y),
        Math.round(crop.w),
        Math.round(crop.h),
        0,
        0,
        Math.round(crop.w),
        Math.round(crop.h)
      );
      URL.revokeObjectURL(url);
      canvas.toBlob(resolve, format, exportQuality);
    };
    img.src = url;
  });

  const nameParts = f.file.name.split('.');
  const baseName = nameParts.slice(0, -1).join('.');
  const resolvedBase = resolveTemplate(filenameTemplate, {
    name: baseName,
    w: crop.w,
    h: crop.h,
    idx: activeIndex + 1,
  });
  const filename = resolvedBase + '.' + ext;

  const a = document.createElement('a');
  a.href = URL.createObjectURL(blob);
  a.download = filename;
  a.click();
  setTimeout(() => URL.revokeObjectURL(a.href), 5000);
  statusText.textContent = `Downloaded: ${filename}`;
}
```
