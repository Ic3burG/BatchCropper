### Step 4: Add preset logic to the script

Just above the `loadSettings();` call at the bottom of the script, add:

```js
// Presets
const PRESETS_KEY = 'batchcrop_presets';

const BUILTIN_PRESETS = [
  { name: 'Square 1:1', ratio: 1 },
  { name: 'Portrait 4:5', ratio: 4 / 5 },
  { name: 'Story 9:16', ratio: 9 / 16 },
  { name: 'Landscape 16:9', ratio: 16 / 9 },
  { name: 'Twitter Header', ratio: 3 / 1 },
  { name: 'Classic 4:3', ratio: 4 / 3 },
];

function loadUserPresets() {
  try {
    const raw = localStorage.getItem(PRESETS_KEY);
    userPresets = raw ? JSON.parse(raw) : [];
  } catch (_) {
    userPresets = [];
  }
}

function saveUserPresetsToStorage() {
  try {
    localStorage.setItem(PRESETS_KEY, JSON.stringify(userPresets));
  } catch (_) {}
}

function applyBuiltInPreset(ratio) {
  if (!imgNatW || !imgNatH) return;
  const imageRatio = imgNatW / imgNatH;
  let w, h;
  if (ratio >= imageRatio) {
    w = imgNatW;
    h = w / ratio;
  } else {
    h = imgNatH;
    w = h * ratio;
  }
  activeCrop.x = (imgNatW - w) / 2;
  activeCrop.y = (imgNatH - h) / 2;
  activeCrop.w = w;
  activeCrop.h = h;
  clampCrop();
  updateInputs();
  updateCropBoxVisual();
  pushHistory();
}

function applyUserPreset(preset) {
  if (!imgNatW) return;
  Object.assign(activeCrop, { x: preset.x, y: preset.y, w: preset.w, h: preset.h });
  clampCrop();
  updateInputs();
  updateCropBoxVisual();
  pushHistory();
}

function renderUserPresets() {
  const container = document.getElementById('user-presets');
  container.replaceChildren();
  userPresets.forEach((preset, i) => {
    const row = document.createElement('div');
    row.className = 'user-preset-row';

    const applyBtn = document.createElement('button');
    applyBtn.className = 'apply';
    applyBtn.textContent = preset.name;
    applyBtn.title =
      Math.round(preset.w) +
      ' x ' +
      Math.round(preset.h) +
      ' at (' +
      Math.round(preset.x) +
      ', ' +
      Math.round(preset.y) +
      ')';
    applyBtn.addEventListener('click', () => applyUserPreset(preset));

    const delBtn = document.createElement('button');
    delBtn.className = 'del';
    delBtn.textContent = 'x';
    delBtn.title = 'Delete preset';
    delBtn.addEventListener('click', () => {
      userPresets.splice(i, 1);
      saveUserPresetsToStorage();
      renderUserPresets();
    });

    row.appendChild(applyBtn);
    row.appendChild(delBtn);
    container.appendChild(row);
  });
}

// Render built-in preset chips
const builtinContainer = document.getElementById('builtin-presets');
BUILTIN_PRESETS.forEach((preset) => {
  const chip = document.createElement('button');
  chip.className = 'preset-chip';
  chip.textContent = preset.name;
  chip.addEventListener('click', () => applyBuiltInPreset(preset.ratio));
  builtinContainer.appendChild(chip);
});

// Save current crop as named preset
document.getElementById('save-preset-btn').addEventListener('click', () => {
  const nameInput = document.getElementById('preset-name-input');
  const name = nameInput.value.trim();
  if (!name || !activeCrop.w || !activeCrop.h) return;
  userPresets.push({ name, x: activeCrop.x, y: activeCrop.y, w: activeCrop.w, h: activeCrop.h });
  saveUserPresetsToStorage();
  renderUserPresets();
  nameInput.value = '';
});

// Collapse/expand presets body
document.getElementById('presets-toggle').addEventListener('click', () => {
  const body = document.getElementById('presets-body');
  const btn = document.getElementById('presets-toggle');
  const collapsed = body.style.display === 'none';
  body.style.display = collapsed ? 'block' : 'none';
  btn.textContent = collapsed ? '\u25be' : '\u25b8';
});

loadUserPresets();
renderUserPresets();
```

function applySortOrder() {
  if (sortOrder === 'none' || !files.length) return;
  const activeFile = files[activeIndex];
  files.sort((a, b) => {
    if (sortOrder === 'name-asc') return a.file.name.localeCompare(b.file.name);
    if (sortOrder === 'name-desc') return b.file.name.localeCompare(a.file.name);
    if (sortOrder === 'size-asc') return a.file.size - b.file.size;
    if (sortOrder === 'size-desc') return b.file.size - a.file.size;
    if (sortOrder === 'status') return (a.done ? 1 : 0) - (b.done ? 1 : 0);
    return 0;
  });
  activeIndex = files.indexOf(activeFile);
  renderFileList();
}

sortSelect.addEventListener('change', (e) => {

### Step 5: Add the applyResizeToCanvas helper function

Add after `clampCrop()` and before `pushHistory()`:

```js
function applyResizeToCanvas(canvas, ctx, img, crop, format) {
  const cropW = Math.round(crop.w);
  const cropH = Math.round(crop.h);
  const cropX = Math.round(crop.x);
  const cropY = Math.round(crop.y);
  const targetW = resizeW || cropW;
  const targetH = resizeH || cropH;

  canvas.width = targetW;
  canvas.height = targetH;

  if (resizeMode === 'fit') {
    const scale = Math.min(targetW / cropW, targetH / cropH);
    const dw = Math.round(cropW * scale);
    const dh = Math.round(cropH * scale);
    const dx = Math.round((targetW - dw) / 2);
    const dy = Math.round((targetH - dh) / 2);
    if (format === 'image/jpeg') {
      ctx.fillStyle = '#000';
      ctx.fillRect(0, 0, targetW, targetH);
    }
    if (flipH || flipV) {
      ctx.translate(flipH ? targetW : 0, flipV ? targetH : 0);
      ctx.scale(flipH ? -1 : 1, flipV ? -1 : 1);
    }
    ctx.drawImage(img, cropX, cropY, cropW, cropH, dx, dy, dw, dh);
  } else if (resizeMode === 'fill') {
    const scale = Math.max(targetW / cropW, targetH / cropH);
    const sw = targetW / scale;
    const sh = targetH / scale;
    const sx = Math.round(cropX + (cropW - sw) / 2);
    const sy = Math.round(cropY + (cropH - sh) / 2);
    if (flipH || flipV) {
      ctx.translate(flipH ? targetW : 0, flipV ? targetH : 0);
      ctx.scale(flipH ? -1 : 1, flipV ? -1 : 1);
    }
    ctx.drawImage(img, sx, sy, Math.round(sw), Math.round(sh), 0, 0, targetW, targetH);
  } else {
    // stretch
    if (flipH || flipV) {
      ctx.translate(flipH ? targetW : 0, flipV ? targetH : 0);
      ctx.scale(flipH ? -1 : 1, flipV ? -1 : 1);
    }
    ctx.drawImage(img, cropX, cropY, cropW, cropH, 0, 0, targetW, targetH);
  }
}
```
