### Step 4: Add `applyWatermark` helper function

Add this function near the other utility functions (e.g., near `applyResizeToCanvas`):

```js
function applyWatermark(canvas, ctx) {
  if (!watermarkEnabled) return;
  if (!watermarkText && !watermarkLogoImg) return;

  const margin = 8;
  const isRight = watermarkCorner.includes('r');
  const isBottom = watermarkCorner.includes('b');

  ctx.save();
  ctx.resetTransform(); // ensure watermark ignores any preceding flip transform
  ctx.globalAlpha = watermarkOpacity;

  let logoH = 0;
  let logoW = 0;

  if (watermarkLogoImg) {
    const maxSize = Math.min(canvas.width, canvas.height) * 0.25;
    const scale = Math.min(
      maxSize / watermarkLogoImg.naturalWidth,
      maxSize / watermarkLogoImg.naturalHeight
    );
    logoW = Math.round(watermarkLogoImg.naturalWidth * scale);
    logoH = Math.round(watermarkLogoImg.naturalHeight * scale);
  }

  const textH = watermarkText ? watermarkFontSize : 0;
  const totalH = logoH + (logoH && textH ? margin : 0) + textH;
  const blockY = isBottom ? canvas.height - totalH - margin : margin;

  if (watermarkLogoImg) {
    const lx = isRight ? canvas.width - logoW - margin : margin;
    ctx.drawImage(watermarkLogoImg, lx, blockY, logoW, logoH);
  }

  if (watermarkText) {
    ctx.font = `${watermarkFontSize}px sans-serif`;
    const textWidth = ctx.measureText(watermarkText).width;
    const tx = isRight ? canvas.width - textWidth - margin : margin;
    const ty = blockY + logoH + (logoH ? margin : 0) + watermarkFontSize;
    ctx.shadowColor = 'rgba(0,0,0,0.7)';
    ctx.shadowBlur = 3;
    ctx.fillStyle = '#ffffff';
    ctx.fillText(watermarkText, tx, ty);
    ctx.shadowBlur = 0;
  }

  ctx.restore();
}
```

**Key note:** `ctx.resetTransform()` is called inside `ctx.save()`/`ctx.restore()` so that preceding flip transforms (from `applyResizeToCanvas` or the normal flip path) do not affect watermark positioning. The transform is restored by `ctx.restore()` but nothing is drawn after the watermark, so this is safe.

### Step 3: Wire the copy listener

Add near other button click listeners:

```js
copyClipboardBtn.addEventListener('click', async () => {
  const f = files[activeIndex];
  if (!f) return;
  const crop = f.crop ?? batchCrop;
  if (!crop.w || !crop.h) return;

  const format = document.getElementById('export-format').value;
  statusText.textContent = `Copying ${f.file.name}...`;

  const blob = await new Promise((resolve) => {
    const img = new Image();
    const url = URL.createObjectURL(f.file);
    img.onload = () => {
      const canvas = document.createElement('canvas');
      canvas.width = Math.round(crop.w);
      canvas.height = Math.round(crop.h);
      const ctx = canvas.getContext('2d');
      if (resizeEnabled && (resizeW || resizeH)) {
        applyResizeToCanvas(canvas, ctx, img, crop, format);
      } else {
        if (flipH || flipV) {
          ctx.translate(flipH ? canvas.width : 0, flipV ? canvas.height : 0);
          ctx.scale(flipH ? -1 : 1, flipV ? -1 : 1);
        }
        ctx.drawImage(
          img,
          Math.round(crop.x),
          Math.round(crop.y),
          Math.round(crop.w),
          Math.round(crop.h),
          0,
          0,
          Math.round(crop.w),
          Math.round(crop.h)
        );
      }
      applyWatermark(canvas, ctx);
      URL.revokeObjectURL(url);
      canvas.toBlob(resolve, 'image/png');
    };
    img.src = url;
  });

  try {
    await navigator.clipboard.write([new ClipboardItem({ 'image/png': blob })]);
    const orig = copyClipboardBtn.textContent;
    copyClipboardBtn.textContent = 'Copied!';
    statusText.textContent = 'Copied to clipboard.';
    setTimeout(() => {
      copyClipboardBtn.textContent = orig;
    }, 2000);
  } catch (err) {
    statusText.textContent = 'Copy failed: ' + err.message;
  }
});
```

Note: always copies as `image/png` (the Clipboard API is most compatible with PNG across browsers). Does **not** set `f.done = true` — copy is non-destructive.
